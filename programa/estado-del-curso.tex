        \parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{%
            \begin{tabularx}{\linewidth}{@{}
                >{\raggedright\arraybackslash}X

                >{\raggedleft\arraybackslash}X
                @{}}
                {\small\scshape\pdfrender{TextRenderingMode=FillStroke,LineWidth=.15pt}sesión actual} &
                {\small\scshape\pdfrender{TextRenderingMode=FillStroke,LineWidth=.15pt}sesión próxima} \\[4pt]
                {\bfseries Semana #1, sesión #2} &
                {\bfseries Semana #4, sesión #5} \\[2pt]
                {\small #3} & {\small #6}

        }%
    }%
    \endgroup\par
}
