% Documento horizontal en papel carta.
\documentclass[10pt,letterpaper]{article}

% Configura el español, conserva el punto decimal
% y evita atajos tipográficos que interfieren con biblatex-apa.
\usepackage[
    spanish,
    es-nodecimaldot,
    es-noshorthands
]{babel}

% Configura la tipografía.
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{microtype}

% Página horizontal con márgenes reducidos.
\usepackage[
    letterpaper,
    landscape,
    margin=1.7cm
]{geometry}

% Paquetes necesarios para la tabla.
\usepackage{array}
\usepackage{longtable}
\usepackage{booktabs}

% Enlaces sin bordes visibles.
\usepackage[hidelinks]{hyperref}

% Datos generales del curso.
\input{../config/curso.tex}

% Indica qué edición está activa.
\input{../config/edicion-activa.tex}

% Incorpora los datos particulares de la edición activa.
\input{../ediciones/\edicionActiva/semestre.tex}

% Permite insertar imágenes.
\usepackage{graphicx}

% Permite diseñar encabezados y pies de página.
\usepackage{fancyhdr}

% Permite usar colores.
\usepackage{xcolor}

% Gestiona correctamente las comillas según el idioma del documento.
\usepackage{csquotes}

% Elimina la sangría de los párrafos.
\setlength{\parindent}{0pt}

% ============================================================
% HEADERS & FOOTERS
% ============================================================

% Elimina cualquier contenido previo del encabezado y del pie.
\fancyhf{}

% Encabezado izquierdo:
% muestra el logo solamente si el archivo existe.
%
% \IfFileExists evita que la compilación falle cuando el logo
% todavía no se ha incorporado al curso.
\fancyhead[L]{%
    \IfFileExists{\rutaLogoInstitucional}{%
        \includegraphics[width=4cm]
        {\rutaLogoInstitucional}%
    }{}%
}

% Encabezado derecho:
% utiliza el texto definido en config/curso.tex.
\fancyhead[R]{%
    \small
    \color{gray}
    \encabezadoInstitucional
}

\renewcommand{\headrulewidth}{0pt}
\setlength{\headheight}{58pt}

\fancyfoot[C]{\thepage}
\pagestyle{fancy}

% ============================================================
% BIBLIOGRAFÍA
% ============================================================

% biblatex administra citas y referencias.
%
% backend=biber:
% usa Biber como procesador bibliográfico.
%
% style=apa:
% aplica el estilo APA.
%
% sorting=nyt:
% ordena por nombre, año y título.
\usepackage[
    backend=biber,
    style=apa,
    sorting=nyt
]{biblatex}

% Indica dónde está la base de datos bibliográfica.
\addbibresource{../bibliografia/referencias.bib}

% ============================================================
% COMANDOS PARA LAS LECTURAS DEL CRONOGRAMA
% ============================================================

% Imprime una cita narrativa conectada con referencias.bib.
%
% #1 es opcional: capítulo, páginas, secciones, etc.
% #2 es obligatorio: clave de la referencia.
%
% Ejemplo:
% \citalectura[cap.~1]{apellido2026}
% produce: Apellido (2026, cap. 1)

\newcommand{\citalectura}[2][]{%
    \textcite[#1]{#2}%
}

% Encabezados reutilizables para clasificar las lecturas.
\newcommand{\lecturasObligatorias}{%
    \textbf{Obligatorias:}\par
}

\newcommand{\lecturaObligatoria}{%
    \textbf{Obligatoria:}\par
}

\newcommand{\lecturasComplementarias}{%
    \par\textbf{Complementarias:}\par
}

\newcommand{\lecturaComplementaria}{%
    \par\textbf{Complementaria:}\par
}

% Mensajes uniformes para sesiones sin lectura.
\newcommand{\sinLecturaAdicional}{%
    \textit{Sin lectura adicional.}%
}

\newcommand{\sinLecturaAsignada}{%
    \textit{Sin lectura asignada.}%
}
